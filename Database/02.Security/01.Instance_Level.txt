Instance level security controls: 
- login
- create database/roles

SUPERUSER - can do anything (NOSUPERUSER)
CREATEDB - make databases
CREATEROLE - make more role
LOGIN - can login into database
REPLICATION - can be used for replication.

# Create user/role§ and assign to generic role
CREATE ROLE accounting NOSUPERUSER NOLOGIN NOCREATEDB;
CREATE ROLE hr NOSUPERUSER NOLOGIN NOCREATEDB;
CREATE USER suzy NOSUPERUSER LOGIN NOCREATEDB password 'pass123';
CREATE USER bobby NOSUPERUSER LOGIN NOCREATEDB password 'pass123';

# See the roles assigned to the specific users
SELECT rolname, rolsuper, rolcreatedb, rolcanlogin
FROM pg_roles
WHERE rolname IN ('hr', 'accounting', 'suzy', 'bobby');

# Check role and its members
SELECT
    r.rolname AS role_name,
    m.rolname AS member_name
FROM
    pg_auth_members am
JOIN pg_roles r ON am.roleid = r.oid
JOIN pg_roles m ON am.member = m.oid
ORDER BY role_name;
